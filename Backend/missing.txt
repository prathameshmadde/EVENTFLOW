app.post(
  '/api/proposals',
  requireAuth,
  requireRole(
    'EVENT_MANAGER'
  ),
  async (
    req,
    res
  ) => {

    try {

      const manager =
        await getManagerProfile(
          req.auth.user_id
        );


      if (
        !manager ||
        upper(
          manager.status
        ) !==
        'ACTIVE'
      ) {

        return res
          .status(403)
          .json({
            message:
              'Active manager profile required'
          });

      }


      const requirementId =
        int(
          req.body.requirement_id ||
          req.body.requirementId
        );


      const packageName =
        text(
          req.body.package_name ||
          req.body.packageName
        );


      const description =
        text(
          req.body.description
        );


      const price =
        num(
          req.body.price
        );


      if (
        !requirementId ||
        !packageName ||
        !price ||
        price <= 0
      ) {

        return res
          .status(400)
          .json({

            message:
              'requirement_id, package_name and price are required'

          });

      }


      const requirement =
        await one(
          `
            SELECT *

            FROM event_requirements

            WHERE
              requirement_id = ?
          `,
          [
            requirementId
          ]
        );


      if (!requirement) {

        return res
          .status(404)
          .json({
            message:
              'Requirement not found'
          });

      }


      if (
        Number(
          requirement.customer_id
        ) ===
        Number(
          req.auth.user_id
        )
      ) {

        return res
          .status(400)
          .json({
            message:
              'You cannot bid on your own requirement'
          });

      }


      if (
        ![
          'OPEN',
          'PROPOSALS_RECEIVED'
        ].includes(
          upper(
            requirement.status
          )
        )
      ) {

        return res
          .status(409)
          .json({
            message:
              'This requirement is no longer accepting proposals'
          });

      }


      const duplicate =
        await one(
          `
            SELECT proposal_id

            FROM proposals

            WHERE

              requirement_id = ?

              AND manager_id = ?

              AND status = 'SUBMITTED'
          `,
          [
            requirementId,
            manager.manager_id
          ]
        );


      if (duplicate) {

        return res
          .status(409)
          .json({
            message:
              'You already have a submitted proposal for this requirement'
          });

      }


      const [
        result
      ] =
        await db.execute(
          `
            INSERT INTO proposals
            (
              requirement_id,
              manager_id,
              package_name,
              description,
              price,
              status
            )

            VALUES
            (
              ?,
              ?,
              ?,
              ?,
              ?,
              ?
            )
          `,
          [

            requirementId,

            manager.manager_id,

            packageName,

            description,

            price,

            'SUBMITTED'

          ]
        );


      await query(
        `
          UPDATE event_requirements

          SET
            status = 'PROPOSALS_RECEIVED'

          WHERE

            requirement_id = ?

            AND status = 'OPEN'
        `,
        [
          requirementId
        ]
      );


      return res
        .status(201)
        .json({

          message:
            '
